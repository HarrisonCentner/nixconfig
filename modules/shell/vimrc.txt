" -----------------------------------------------------------------------------
"  CoC.Vim
"  ----------------------------------------------------------------------------

" Use <space> as leader
let mapleader = " "

" === Coc basic keybindings ===

" Show documentation (hover)
nnoremap <silent> K :call CocActionAsync('doHover')<CR>

" Jump to definition
nnoremap <silent> gd <Plug>(coc-definition)

" Jump to type definition
nnoremap <silent> gy <Plug>(coc-type-definition)

" Jump to implementation
nnoremap <silent> gi <Plug>(coc-implementation)

" Jump to references
nnoremap <silent> gr <Plug>(coc-references)

" Go back to last jump location
nnoremap <silent> <C-o> <C-o>
" (This is just Vim’s built-in jump back, works after gd/gr/etc.)

" Rename symbol
nnoremap <leader>rn <Plug>(coc-rename)

" Format selected code
xnoremap <leader>f  <Plug>(coc-format-selected)
nnoremap <leader>f  <Plug>(coc-format-selected)

" Show diagnostics
nnoremap <silent> [g <Plug>(coc-diagnostic-prev)
nnoremap <silent> ]g <Plug>(coc-diagnostic-next)

" Code actions (like quick fix)
nnoremap <leader>ac  <Plug>(coc-codeaction)
xnoremap <leader>ac  <Plug>(coc-codeaction-selected)

" Always scroll on floating docs if they are open
nnoremap <silent><nowait><expr> <C-f> coc#float#has_scroll() ? coc#float#scroll(1) : "\<C-f>"
nnoremap <silent><nowait><expr> <C-b> coc#float#has_scroll() ? coc#float#scroll(0) : "\<C-b>"
inoremap <silent><nowait><expr> <C-f> coc#float#has_scroll() ? "\<c-r>=coc#float#scroll(1)\<cr>" : "\<Right>"
inoremap <silent><nowait><expr> <C-b> coc#float#has_scroll() ? "\<c-r>=coc#float#scroll(0)\<cr>" : "\<Left>"
vnoremap <silent><nowait><expr> <C-f> coc#float#has_scroll() ? coc#float#scroll(1) : "\<C-f>"
vnoremap <silent><nowait><expr> <C-b> coc#float#has_scroll() ? coc#float#scroll(0) : "\<C-b>"

" Highlight symbol under cursor
autocmd CursorHold * silent call CocActionAsync('highlight')

" Tab completion
inoremap <silent><expr> <TAB> coc#pum#visible() ? coc#pum#confirm() : "\<C-g>u\<TAB>"
